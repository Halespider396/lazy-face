cmake_minimum_required(VERSION 3.16)
project(lazy-face CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui videoio objdetect dnn)
add_executable(lazy-face main.cpp)
target_link_libraries(lazy-face PRIVATE opencv_core opencv_imgproc opencv_highgui opencv_videoio opencv_objdetect opencv_dnn)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
target_include_directories(lazy-face PRIVATE ${OpenCV_INCLUDE_DIRS})
